#### Extract all detected experimental barcodes and their frequencies (x = length of UMI1 = 5, y = length of the experimental barcodes = 6 )

zcat iClip_S1_R1_001.fastq.gz | awk -v umi1_len=5 -v exp_bc_len=6 '{ if (FNR%4==2) print substr($1,(umi1_len+1),exp_bc_len) }' | sort | uniq -c | sort -k1,1rn > exp_barcodes_R1.detected



  
  
#### Quality control of barcode region of R1 (Option A: barcode-only, 6nt)
###------------------------
mkdir -p tmp_R1

zcat iClip_S1_R1_001.fastq.gz | \
  fastx_trimmer -f 1 -l 6 | \
  fastq_quality_filter -q 10 -p 100 | \
  awk 'FNR%4==1 { print substr($1,2) }' > tmp_R1/iClip_S1_R1_001.qualFilteredIDs.list

seqtk subseq iClip_S1_R1_001.fastq.gz tmp_R1/iClip_S1_R1_001.qualFilteredIDs.list | \
  sed 's/ /#/g; s/\//#/g' | gzip > iClip_S1_R1_001.filtered.fastq.gz

#### equalise number of reads in both R1 and R2
###------------------------
export LC_ALL=C

zcat iClip_S1_R1_001.filtered.fastq.gz | awk 'NR%4==1{n=split($0,a,"#"); print substr(a[1],2)}' | sort > R1_read_ids.txt
zcat iClip_S1_R2_002.fastq.gz         | awk 'NR%4==1{n=split($0,a,"#"); print substr(a[1],2)}' | sort > R2_read_ids.txt
